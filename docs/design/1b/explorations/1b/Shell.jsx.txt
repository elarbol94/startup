const ThemeCtx = React.createContext({ mode: "Dunkel", setMode: () => {} });
const isDark = (mode) => mode === "Dunkel" || (mode === "System" && matchMedia("(prefers-color-scheme: dark)").matches);
function ThemeToggle() {
  const { mode, setMode } = React.useContext(ThemeCtx);
  const dark = isDark(mode);
  return <button className="btn btn-g btn-ic btn-sm" title={dark ? "Hell" : "Dunkel"} aria-label="Farbmodus wechseln" onClick={() => setMode(dark ? "Hell" : "Dunkel")}><I n={dark ? "Sun" : "Moon"} /></button>;
}

const SCREENS = () => ({ overview: OverviewScreen, calendar: CalendarScreen, projects: ProjectsScreen, time: TimeScreen, meetings: MeetingsScreen, accounting: AccountingScreen1b, personnel: PersonnelScreen, wiki: WikiScreen, municipalities: MunicipalitiesScreen, network: NetworkScreen, settings: SettingsScreen });
const load = (k, d) => { try { const v = JSON.parse(localStorage.getItem(k)); return v ?? d; } catch (e) { return d; } };

function App1b() {
  if (!NS1B || !NS1B.GroupedSidebar || !NS1B.WorkspaceTabs || !NS1B.CommandPalette) return <div style={{ padding: 40, fontFamily: "var(--font-sans)", color: "var(--foreground)" }}>Die Komponentenbibliothek ist noch nicht aktuell. Bitte Seite neu laden.</div>;
  return <App1bInner />;
}
function App1bInner() {
  const qp = new URLSearchParams(location.search), embed = qp.has("screen") && ALL[qp.get("screen")];
  const saved = embed ? { tabs: ["overview", qp.get("screen")].filter((v, i, a) => a.indexOf(v) === i), cur: qp.get("screen") } : load("mp-1b-tabs", {});
  const [tabs, setTabs] = React.useState(saved.tabs && saved.tabs.length ? saved.tabs : ["overview", "projects"]);
  const [cur, setCur] = React.useState(saved.cur || "overview");
  const [mode, setModeRaw] = React.useState(() => qp.get("mode") || localStorage.getItem("mp-1b-theme") || "Dunkel");
  const [collapsed, setCollapsed] = React.useState(() => load("mp-1b-collapsed", false));
  const [search, setSearch] = React.useState(false);
  const [drawer, setDrawer] = React.useState(qp.get("drawer") === "1");
  const setMode = (m) => { setModeRaw(m); if (!embed) localStorage.setItem("mp-1b-theme", m); };
  const [dataMode, setDataModeRaw] = React.useState(() => embed ? (qp.get("data") || "Daten") : (localStorage.getItem("mp-1b-data") || "Daten"));
  const setDataMode = (m) => { setDataModeRaw(m); if (!embed) localStorage.setItem("mp-1b-data", m); };
  React.useEffect(() => { document.documentElement.classList.toggle("dark", isDark(mode)); }, [mode]);
  React.useEffect(() => { if (!embed) localStorage.setItem("mp-1b-tabs", JSON.stringify({ tabs, cur })); }, [tabs, cur]);
  React.useEffect(() => { localStorage.setItem("mp-1b-collapsed", JSON.stringify(collapsed)); }, [collapsed]);
  React.useEffect(() => {
    const k = (e) => { if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "k") { e.preventDefault(); setSearch(true); } };
    document.addEventListener("keydown", k); return () => document.removeEventListener("keydown", k);
  }, []);
  const toast = (t) => NS1B.Toaster.toast.info(t);
  const go = (k) => { setTabs((ts) => ts.includes(k) ? ts : [...ts, k]); setCur(k); setDrawer(false); };
  const close = (k) => { const next = tabs.filter((t) => t !== k); const list = next.length ? next : ["overview"]; setTabs(list); if (cur === k) setCur(list[Math.max(0, tabs.indexOf(k) - 1)] || list[0]); };
  const cmdItems = [
    ...["Neue Aufgabe", "Neuer Termin", "Neue Deadline", "Neuer Kontakt"].map((l, i) => ({ group: "Erstellen", label: l, icon: ["ClipboardPlus", "CalendarPlus", "CalendarClock", "UserPlus"][i], hint: ["N", "", "D", ""][i], onSelect: () => toast(l + " – Formular würde sich öffnen.") })),
    ...Object.values(ALL).map((x) => ({ group: "Navigation", label: x.l, icon: x.i, onSelect: () => go(x.k) })),
    ...["Projektbeschreibung_V1", "Digitalisierung in österreichischen Gemeinden", "Mail-Hosting"].map((l) => ({ group: "Wiki", label: l, icon: "FileText", hint: "Dokument", onSelect: () => go("wiki") })),
    ...["BANOND Wireframe", "Elevator-Pitch Einreichung", "Portainer einrichten"].map((l) => ({ group: "Aufgaben", label: l, icon: "ClipboardCheck", onSelect: () => go("overview") })),
    ...["Daten", "Leer", "Laden"].map((m) => ({ group: "Demo-Zustand", label: "Zustand: " + m, icon: { Daten: "Database", Leer: "Inbox", Laden: "Loader" }[m], keywords: "empty loading leer laden", onSelect: () => setDataMode(m) })),
    { group: "Einstellungen", label: "Farbmodus wechseln", icon: "SunMoon", keywords: "hell dunkel theme", onSelect: () => setMode(isDark(mode) ? "Hell" : "Dunkel") },
  ];
  const Screen = SCREENS()[cur];
  return (
    <ThemeCtx.Provider value={{ mode, setMode }}><DataCtx.Provider value={{ mode: dataMode, setMode: setDataMode }}>
      <div className="app">
        <Nav cur={cur} go={go} collapsed={collapsed} onToggle={() => setCollapsed(!collapsed)} onSearch={() => setSearch(true)} toast={toast} style={{ height: "100vh" }} />
        <div className="main">
          <header className="mhdr">
            <button className="btn btn-g btn-ic" aria-label="Navigation öffnen" onClick={() => setDrawer(true)}><I n="Menu" s={20} /></button>
            <b className="ell grow" style={{ fontSize: 14 }}>{ALL[cur].tab || ALL[cur].l}</b>
            <button className="btn btn-g btn-ic" aria-label="Suchen" onClick={() => setSearch(true)}><I n="Search" s={20} /></button>
            <Av p="a" size={30} />
          </header>
          <NS1B.WorkspaceTabs tabs={tabs.map((k) => ({ id: k, label: ALL[k].tab || ALL[k].l, icon: ALL[k].i }))} activeId={cur} onSelect={setCur} onClose={close} onNew={() => setSearch(true)} onSplit={() => toast("Geteilte Ansicht – nicht nachgebaut.")} />
          <div className="page" key={cur + dataMode} data-screen-label={ALL[cur].l}>{dataMode === "Laden" && cur !== "settings" ? <SkeletonPage /> : <Screen go={go} />}</div>
        </div>
        {drawer ? <div className="drawer" onMouseDown={(e) => { if (e.target === e.currentTarget) setDrawer(false); }}><Nav cur={cur} go={go} collapsed={false} onSearch={() => { setDrawer(false); setSearch(true); }} toast={toast} style={{ width: "min(288px, 86vw)", height: "100%" }} /></div> : null}
        <NS1B.CommandPalette open={search} onOpenChange={setSearch} items={cmdItems} placeholder="Module, Dokumente, Aufgaben oder Befehle suchen…" />
        {dataMode !== "Daten" ? <div style={{ position: "fixed", right: 16, bottom: 16, zIndex: 40, display: "flex", alignItems: "center", gap: 8, padding: "6px 6px 6px 12px", borderRadius: 999, background: "var(--popover)", border: "1px solid var(--border)", boxShadow: "var(--shadow-lg)", fontSize: 12.5 }}><I n={dataMode === "Leer" ? "Inbox" : "Loader"} s={14} />Demo-Zustand: {dataMode}<button className="btn btn-sm btn-o" onClick={() => setDataMode("Daten")}>Mit Daten</button></div> : null}
        <NS1B.Toaster />
      </div>
    </DataCtx.Provider></ThemeCtx.Provider>
  );
}
