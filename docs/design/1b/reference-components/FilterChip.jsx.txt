import React from "react";
import { Icon } from "../core/Icon";
const cx = (...a) => a.filter(Boolean).join(" ");

export function FilterChip({ active = false, onRemove, removeLabel = "Filter entfernen", className, children, ...props }) {
  return (
    <span data-active={active} className={cx("mp-chip", className)} {...props}>
      {children}
      {onRemove ? <button type="button" className="mp-chip__x" aria-label={removeLabel} onClick={onRemove}><Icon name="X" size={11} /></button> : null}
    </span>
  );
}
