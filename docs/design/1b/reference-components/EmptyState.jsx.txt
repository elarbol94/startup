import React from "react";
import { Icon } from "../core/Icon";
const cx = (...a) => a.filter(Boolean).join(" ");
/** Short empty state: icon tile, one-line title, one sentence, one or two actions. */
export function EmptyState({ icon = "Inbox", title, description, actions, compact = false, className, style }) {
  return (
    <div className={cx("mp-empty", compact && "mp-empty--compact", className)} style={style}>
      <span className="mp-empty__icon"><Icon name={icon} size={18} /></span>
      <div className="mp-empty__title">{title}</div>
      {description ? <p className="mp-empty__desc">{description}</p> : null}
      {actions ? <div className="mp-empty__actions">{actions}</div> : null}
    </div>
  );
}
