import React from "react";
import { Icon } from "../core/Icon";
const cx = (...a) => a.filter(Boolean).join(" ");

export function SegmentedControl({ options = [], value, defaultValue, onValueChange, size = "default", className, ...props }) {
  const [inner, setInner] = React.useState(defaultValue ?? (options[0] && options[0].value));
  const cur = value === undefined ? inner : value;
  return (
    <div role="radiogroup" className={cx("mp-seg", size === "sm" && "mp-seg--sm", className)} {...props}>
      {options.map((o) => (
        <button key={o.value} type="button" role="radio" aria-checked={o.value === cur} aria-label={o.label ? undefined : o.value} data-active={o.value === cur} className="mp-seg__opt"
          onClick={() => { if (value === undefined) setInner(o.value); onValueChange && onValueChange(o.value); }}>
          {o.icon ? <Icon name={o.icon} size={14} /> : null}{o.label}
        </button>
      ))}
    </div>
  );
}
